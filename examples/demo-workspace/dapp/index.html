SilverScript source contracts/auction.sil 
contract KaspaAuction(pubkey seller, pubkey highestBidder, int minBidSompi) {
  entry close(sig sellerSig) {
    require(checkSig(sellerSig, seller));
  }
  entry claim(sig winnerSig) {
    require(checkSig(winnerSig, highestBidder));
  }
  entry cancel(sig sellerSig) {
    require(checkSig(sellerSig, seller));
  }
}
